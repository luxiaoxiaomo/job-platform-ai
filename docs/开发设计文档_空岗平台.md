## 1. 快速开始（5分钟）

### 1.1 克隆项目（假设已创建）

```powershell
# 克隆代码
git clone <repository-url>
cd job-platform

# 安装依赖
poetry install

# 复制环境变量
copy .env.example .env
# 编辑 .env，填入你的配置
```

### 1.2 启动数据库

```powershell
# 方式1：Docker Compose（推荐）
docker-compose -f docker-compose.dev.yml up -d postgres redis

# 方式2：本地已安装PostgreSQL和Redis
# 确保PostgreSQL在运行（默认端口5432）
# 确保Redis在运行（默认端口6379）

# 创建数据库
psql -U postgres
CREATE DATABASE jobplatform_dev;
CREATE EXTENSION vector;
\q
```

### 1.3 初始化数据库

```powershell
# 运行数据库迁移
poetry run alembic upgrade head

# （可选）导入测试数据
poetry run python scripts/seed_data.py
```

### 1.4 启动应用

```powershell
# 启动FastAPI（带热重载）
poetry run uvicorn app.main:app --reload --port 8000

# 打开浏览器
# http://localhost:8000/docs  ← Swagger API文档
# http://localhost:8000/redoc ← ReDoc API文档
```

### 1.5 验证安装

```powershell
# 测试API
curl http://localhost:8000/health

# 应该返回
# {"status": "ok"}
```

**🎉 恭喜！开发环境已就绪！**

## 4. 代码规范

### 4.1 命名规范

**文件命名**：
- Python文件：`snake_case.py`（例：`job_service.py`）
- 配置文件：`kebab-case.yml`（例：`docker-compose.dev.yml`）

**变量命名**：
```python
# ✅ 推荐
user_id = 123
job_title = "Python开发工程师"
ai_result = await gateway.call()

# ❌ 不推荐
userId = 123           # camelCase（Java风格）
JobTitle = "..."       # PascalCase（应该用于类名）
ai_Result = ...        # 混合风格
```

**函数命名**：
```python
# ✅ 推荐：动词开头，描述行为
def create_job(data: JobCreate) -> Job:
    pass

def get_user_by_id(user_id: int) -> User:
    pass

async def send_push_notification(user_id: int, message: str):
    pass

# ❌ 不推荐
def job(data):  # 太简短
    pass

def CreateJob(data):  # PascalCase
    pass
```

**类命名**：
```python
# ✅ 推荐：PascalCase
class UserService:
    pass

class JobRepository:
    pass

class AIGateway:
    pass

# ❌ 不推荐
class user_service:  # snake_case
    pass
```

**常量命名**：
```python
# ✅ 推荐：全大写+下划线
MAX_FILE_SIZE = 10 * 1024 * 1024  # 10MB
AI_MONTHLY_BUDGET = 5000
DEFAULT_PAGE_SIZE = 20

# ❌ 不推荐
maxFileSize = 10 * 1024 * 1024
```

### 4.2 类型注解（强制）

```python
# ✅ 推荐：所有函数都要类型注解
from typing import List, Optional

def get_jobs(

    pass

async def create_user(data: UserCreate) -> User:
    pass

# ❌ 不推荐：缺少类型注解
def get_jobs(city=None, limit=10):
    pass
```

#### Step 2：定义数据模型（models.py）

```python
# app/modules/interview/models.py
from sqlalchemy import Column, Integer, String, DateTime, ForeignKey, Enum as SQLEnum
from sqlalchemy.dialects.postgresql import JSONB
from app.db.base import Base
from datetime import datetime

class Interview(Base):
    __tablename__ = "interviews"
    
    id = Column(Integer, primary_key=True, index=True)
    job_id = Column(Integer, ForeignKey("jobs.id"), nullable=False)
    recruiter_id = Column(Integer, ForeignKey("users.id"), nullable=False)
    candidate_id = Column(Integer, ForeignKey("users.id"), nullable=False)
    
    scheduled_at = Column(DateTime, nullable=False)
    location = Column(String(200))
    interview_type = Column(SQLEnum('online', 'onsite', 'phone', name='interview_type'))
    
    status = Column(
        SQLEnum('scheduled', 'completed', 'cancelled', name='interview_status'),
        default='scheduled'
    )
    
    feedback = Column(JSONB)  # AI生成的面试反馈
    result = Column(SQLEnum('pass', 'fail', 'pending', name='interview_result'))
    
    created_at = Column(DateTime, default=datetime.utcnow)
    updated_at = Column(DateTime, default=datetime.utcnow, onupdate=datetime.utcnow)
```

#### Step 4：实现Repository（repository.py）

```python
# app/modules/interview/repository.py
from sqlalchemy import select
from sqlalchemy.ext.asyncio import AsyncSession
from typing import List, Optional
from app.modules.interview.models import Interview

class InterviewRepository:
    """面试数据访问层"""
    
    @staticmethod
    async def create(db: AsyncSession, interview: Interview) -> Interview:
        """创建面试"""
        db.add(interview)
        await db.commit()
        await db.refresh(interview)
        return interview
    
    @staticmethod
    async def get_by_id(db: AsyncSession, interview_id: int) -> Optional[Interview]:
        """根据ID查询"""
        result = await db.execute(
            select(Interview).where(Interview.id == interview_id)
        )
        return result.scalar_one_or_none()
    
    @staticmethod
    async def get_by_recruiter(
        db: AsyncSession,
        recruiter_id: int,
        status: Optional[str] = None
    ) -> List[Interview]:
        """查询招聘者的面试列表"""
        query = select(Interview).where(Interview.recruiter_id == recruiter_id)
        
        if status:
            query = query.where(Interview.status == status)
        
        query = query.order_by(Interview.scheduled_at.desc())
        result = await db.execute(query)
        return result.scalars().all()
    
    @staticmethod
    async def update(db: AsyncSession, interview: Interview) -> Interview:
        """更新面试"""
        await db.commit()
        await db.refresh(interview)
        return interview
```

#### Step 5：实现Service（service.py）

```python
# app/modules/interview/service.py
from sqlalchemy.ext.asyncio import AsyncSession
from app.modules.interview.models import Interview
from app.modules.interview.schemas import InterviewCreate, InterviewUpdate
from app.modules.interview.repository import InterviewRepository
from app.ai.gateway import ai_gateway
from app.tasks.push_tasks import send_interview_notification
from fastapi import HTTPException

class InterviewService:
    """面试业务逻辑层"""
    
    @staticmethod
    async def create_interview(
        db: AsyncSession,
        data: InterviewCreate,
        recruiter_id: int
    ) -> Interview:
        """创建面试并发送通知"""
        
        # 1. 创建面试记录
        interview = Interview(
            job_id=data.job_id,
            recruiter_id=recruiter_id,
            candidate_id=data.candidate_id,
            scheduled_at=data.scheduled_at,
            location=data.location,
            interview_type=data.interview_type
        )
        
        interview = await InterviewRepository.create(db, interview)
        
        # 2. 异步发送通知（Celery任务）
        send_interview_notification.delay(interview.id)
        
        return interview
    
    @staticmethod
    async def record_feedback(
        db: AsyncSession,
        interview_id: int,
        feedback_text: str,
        result: str
    ) -> Interview:
        """记录面试反馈（AI辅助）"""
        
        # 1. 查询面试
        interview = await InterviewRepository.get_by_id(db, interview_id)
        if not interview:
            raise HTTPException(status_code=404, detail="面试不存在")
        
        # 2. AI分析反馈（结构化）
        ai_feedback = await ai_gateway.call_with_fallback(
            chain="analyze_interview_feedback",
            inputs={"feedback": feedback_text},
            feature="interview_feedback"
        )
        
        # 3. 更新面试记录
        interview.feedback = ai_feedback
```

```python
async def get_jobs_paginated(
    db: AsyncSession,
    page: int = 1,
    page_size: int = 20
) -> List[Job]:
    """分页查询"""
    offset = (page - 1) * page_size
    
    result = await db.execute(
        select(Job)
        .where(Job.status == 'online')
        .order_by(Job.created_at.desc())
        .limit(page_size)
```

## 12. 常见问题FAQ

### Q1：数据库连接池耗尽

**问题**：
```
sqlalchemy.exc.TimeoutError: QueuePool limit of size 5 overflow 10 reached
```

**原因**：数据库连接未正确关闭

**解决**：
```python
# ✅ 正确：使用依赖注入，自动管理连接
@router.get("/jobs")
async def get_jobs(db: AsyncSession = Depends(get_db)):
    # db会自动关闭
    pass

# ❌ 错误：手动创建连接，忘记关闭
async def get_jobs():
    db = SessionLocal()
    # ... 忘记 await db.close()
```

### Q2：Pydantic验证错误

**问题**：
```
ValidationError: 1 validation error for JobCreate
cities
  field required
```

**原因**：请求数据缺少必填字段

**解决**：
```python
# 检查Schema定义
class JobCreate(BaseModel):
    cities: List[str] = Field(..., min_items=1)  # 必填
    # 改为可选
    cities: Optional[List[str]] = None
```

### Q3：异步函数未await

**问题**：
```
RuntimeWarning: coroutine 'create_job' was never awaited
```

**原因**：调用async函数时忘记await

**解决**：
```python
# ❌ 错误
job = create_job(data)

# ✅ 正确
job = await create_job(data)
```

### Q4：数据库迁移冲突

**问题**：
```
alembic.util.CommandError: Multiple head revisions are present
```

**原因**：多人同时创建迁移脚本

**解决**：
```powershell
# 合并head
poetry run alembic merge heads -m "merge heads"
poetry run alembic upgrade head
```

### Q5：AI调用超时

**问题**：OpenAI API调用超过30秒

**解决**：
```python
# 增加超时时间
llm = ChatOpenAI(model="gpt-4o-mini", timeout=60)

# 或启用流式响应
llm = ChatOpenAI(model="gpt-4o-mini", streaming=True)
```

### Q6：Redis连接失败

**问题**：
```
redis.exceptions.ConnectionError: Error 10061 connecting to localhost:6379
```

**解决**：
```powershell
# 确认Redis正在运行
redis-cli ping

# 如果使用WSL
wsl
sudo service redis-server start
```

## 14. 性能优化指南

### 14.1 数据库优化

#### 使用连接池

```python
# app/db/session.py
from sqlalchemy.ext.asyncio import create_async_engine, AsyncSession
from sqlalchemy.orm import sessionmaker

engine = create_async_engine(
    DATABASE_URL,
    echo=False,
    pool_size=10,          # 连接池大小
    max_overflow=20,       # 超出pool_size的最大连接数
    pool_timeout=30,       # 获取连接超时时间
    pool_recycle=3600,     # 连接回收时间（1小时）
    pool_pre_ping=True     # 连接前测试（避免断连）
)
```

#### 批量操作

```python
# ❌ 错误：循环插入（N次数据库调用）
for data in job_list:
    job = Job(**data)
    db.add(job)
    await db.commit()

# ✅ 正确：批量插入（1次数据库调用）
jobs = [Job(**data) for data in job_list]
db.add_all(jobs)
await db.commit()
```

#### 使用select in loading

```python
# 避免N+1查询
from sqlalchemy.orm import selectinload

jobs = await db.execute(
    select(Job)
    .options(selectinload(Job.enterprise))  # 预加载
    .limit(10)
)
```

### 14.2 缓存策略

```python
from functools import lru_cache
from app.core.cache import redis_client

# 1. 内存缓存（适合不变的数据）
@lru_cache(maxsize=128)
def get_city_list():
    return ["北京", "上海", "深圳", ...]

# 2. Redis缓存（适合热点数据）
async def get_job_cached(job_id: int):
    cache_key = f"job:{job_id}"
    
    # 先查缓存
    cached = await redis_client.get(cache_key)
    if cached:
        return json.loads(cached)
    
    # 查数据库
    job = await JobRepository.get_by_id(db, job_id)
    
    # 写缓存
    await redis_client.setex(
        cache_key,
        600,  # 10分钟
        json.dumps(job.dict())
    )
    
    return job

# 3. 缓存穿透保护
async def get_with_cache_protection(key: str):
    # 使用空值占位，避免缓存穿透
    cached = await redis_client.get(key)
    if cached == "NULL":
        return None
    if cached:
        return json.loads(cached)
    
    data = await fetch_from_db(key)
    if data is None:
        await redis_client.setex(key, 60, "NULL")
    else:
        await redis_client.setex(key, 600, json.dumps(data))
    
    return data
```

### 14.3 异步优化
